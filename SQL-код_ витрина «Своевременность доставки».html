Схема исходных таблиц

sql/01_schema.sql
-- 01_schema.sql — исходные таблицы (PostgreSQL). Даты в CSV имеют формат дд.мм.гггг.
SET datestyle = 'ISO, DMY';

DROP TABLE IF EXISTS delivery_times, orders, delivery_status, departments CASCADE;

CREATE TABLE departments (
    department_id   integer PRIMARY KEY,
    department_name varchar(255)            -- у отделений 199 и 200 название в источнике пустое
);

CREATE TABLE delivery_status (
    status_id integer PRIMARY KEY,           -- 1 delivered, 2 declined, 3 in_transit
    status    varchar(255) NOT NULL
);

-- ВАЖНО: order_id в источнике НЕ уникален: 57 номеров встречаются дважды у разных отделений,
-- а для двух пар (362381/37 и 375904/187) совпадает даже отделение. Поэтому ключ заказа в витрине —
-- пара (order_id, department_id), а неоднозначные пары исключаются (см. 03_mart_delivery_ontime.sql).
-- В «сырых» таблицах первичный ключ по этой паре поставить нельзя.
CREATE TABLE orders (
    order_id      integer NOT NULL,
    department_id integer NOT NULL REFERENCES departments,
    order_date    date    NOT NULL,
    delivery_date date,                      -- NULL, если не доставлен (в пути или отказ)
    status_id     integer NOT NULL REFERENCES delivery_status
);
CREATE INDEX ON orders (order_id, department_id);

CREATE TABLE delivery_times (
    department_id integer NOT NULL REFERENCES departments,
    order_id      integer NOT NULL,
    deadline_date date    NOT NULL           -- крайний срок доставки
);
CREATE INDEX ON delivery_times (order_id, department_id);

Витрина mart_delivery_ontime (основной SQL-запрос)

sql/03_mart_delivery_ontime.sql
-- 03_mart_delivery_ontime.sql
-- Витрина «Своевременность доставки»: одна строка = отделение × день отправки (order_date)
-- Период: с 01.01.2021. Показатели: отправлено, доставлено (вовремя/с опозданием), отказы, в пути, % вовремя.
--
-- Бизнес-правила:
--   * день витрины — дата заказа/отправления из отделения (order_date);
--   * «вовремя» = заказ доставлен (status_id = 1) и delivery_date <= deadline_date;
--   * on_time_pct = доставлено вовремя / доставлено (в знаменателе только доставленные:
--     заказы «в пути» ещё не оценить, «отказы» — не вина срока доставки, их доля вынесена отдельно);
--   * строки есть только для дней, когда из отделения были отправления (пустые дни BI заполняет сам).
--   * order_id в источнике не уникален, поэтому заказы соединяются по паре (order_id, department_id);
--     2 пары (4 заказа из 9999) повторяются даже внутри отделения с разными датами, а в delivery_times
--     нет даты заказа, чтобы однозначно подобрать срок — они исключены из витрины.

DROP TABLE IF EXISTS mart_delivery_ontime;

CREATE TABLE mart_delivery_ontime AS
WITH uniq_orders AS (      -- пары, которые однозначно соединяются с delivery_times
    SELECT order_id, department_id FROM orders         GROUP BY 1, 2 HAVING count(*) = 1
    INTERSECT
    SELECT order_id, department_id FROM delivery_times GROUP BY 1, 2 HAVING count(*) = 1
), base AS (
    SELECT o.order_date,
           o.department_id,
           o.status_id,
           o.delivery_date,
           t.deadline_date
    FROM orders o
    JOIN delivery_times t
      ON t.order_id = o.order_id
     AND t.department_id = o.department_id
    JOIN uniq_orders u
      ON u.order_id = o.order_id
     AND u.department_id = o.department_id
    WHERE o.order_date >= DATE '2021-01-01'
), as_of AS (
    SELECT max(order_date) AS as_of_date FROM orders      -- «сегодня» для данных выгрузки
)
SELECT b.order_date                                                       AS report_date,
       b.department_id,
       COALESCE(d.department_name, 'Отделение №' || b.department_id)      AS department_name,
       count(*)                                                           AS orders_total,
       count(*) FILTER (WHERE b.status_id = 1)                            AS delivered_cnt,
       count(*) FILTER (WHERE b.status_id = 1 AND b.delivery_date <= b.deadline_date) AS delivered_on_time_cnt,
       count(*) FILTER (WHERE b.status_id = 1 AND b.delivery_date >  b.deadline_date) AS delivered_late_cnt,
       count(*) FILTER (WHERE b.status_id = 2)                            AS declined_cnt,
       count(*) FILTER (WHERE b.status_id = 3)                            AS in_transit_cnt,
       count(*) FILTER (WHERE b.status_id = 3 AND b.deadline_date < a.as_of_date) AS in_transit_overdue_cnt,
       round(100.0 * count(*) FILTER (WHERE b.status_id = 1 AND b.delivery_date <= b.deadline_date)
             / NULLIF(count(*) FILTER (WHERE b.status_id = 1), 0), 1)     AS on_time_pct
FROM base b
CROSS JOIN as_of a
JOIN departments d ON d.department_id = b.department_id
GROUP BY b.order_date, b.department_id, d.department_name
ORDER BY b.order_date, b.department_id;

ALTER TABLE mart_delivery_ontime ADD PRIMARY KEY (report_date, department_id);
COMMENT ON TABLE mart_delivery_ontime IS 'Витрина: своевременность доставки по отделениям и дням (с 2021-01-01)';

Запросы карточек дашборда Metabase

sql/04_metabase_cards.sql
-- 04_metabase_cards.sql — запросы карточек дашборда Metabase (источник — витрина mart_delivery_ontime)
-- Переменные — Field Filter:
--   {{department}} → mart_delivery_ontime.department_name (Dropdown list)
--   {{report_date}} → mart_delivery_ontime.report_date    (Date: All options / диапазон)
-- Важно: % вовремя считаем как sum(вовремя)/sum(доставлено), а НЕ как среднее по дневным процентам.

-- === Карточка 1. Доставлено вовремя, % (Number) ===
SELECT round(100.0 * sum(delivered_on_time_cnt) / NULLIF(sum(delivered_cnt), 0), 1) AS "Вовремя, %"
FROM mart_delivery_ontime
WHERE 1=1 [[AND {{department}}]] [[AND {{report_date}}]];

-- === Карточка 2. Отправлений за период (Number) ===
SELECT sum(orders_total) AS "Отправлений"
FROM mart_delivery_ontime
WHERE 1=1 [[AND {{department}}]] [[AND {{report_date}}]];

-- === Карточка 3. Сейчас в пути (Number; период не учитывается — это состояние «на данный момент») ===
SELECT sum(in_transit_cnt) AS "В пути", sum(in_transit_overdue_cnt) AS "из них просрочено"
FROM mart_delivery_ontime
WHERE 1=1 [[AND {{department}}]];

-- === Карточка 4. Своевременность по месяцам (Line; X = month, Y = on_time_pct) ===
SELECT date_trunc('month', report_date)::date AS month,
       round(100.0 * sum(delivered_on_time_cnt) / NULLIF(sum(delivered_cnt), 0), 1) AS on_time_pct
FROM mart_delivery_ontime
WHERE 1=1 [[AND {{department}}]] [[AND {{report_date}}]]
GROUP BY 1
ORDER BY 1;

-- === Карточка 5. Отправления по месяцам (Bar; X = month, Y = orders) ===
SELECT date_trunc('month', report_date)::date AS month,
       sum(orders_total) AS orders
FROM mart_delivery_ontime
WHERE 1=1 [[AND {{department}}]] [[AND {{report_date}}]]
GROUP BY 1
ORDER BY 1;

-- === Карточка 6. Отправления в пути по отделениям (Table) ===
SELECT department_name       AS "Отделение",
       sum(in_transit_cnt)         AS "В пути",
       sum(in_transit_overdue_cnt) AS "Просрочено",
       min(report_date) FILTER (WHERE in_transit_cnt > 0) AS "Самый старый заказ"
FROM mart_delivery_ontime
WHERE 1=1 [[AND {{department}}]]
GROUP BY department_name
HAVING sum(in_transit_cnt) > 0
ORDER BY 3 DESC, 2 DESC
LIMIT 20;

-- === Карточка 7. Чем заканчиваются отправления (Bar; X = outcome, Y = orders) ===
-- Рейтинг отделений по % вовремя намеренно не строим: при ~27 доставках на отделение
-- различия между ними статистически неотличимы от случайных (хи-квадрат, p = 0,24).
SELECT outcome, orders FROM (
  SELECT 1 AS ord, 'Доставлено вовремя' AS outcome, sum(delivered_on_time_cnt) AS orders
    FROM mart_delivery_ontime WHERE 1=1 [[AND {{department}}]] [[AND {{report_date}}]]
  UNION ALL
  SELECT 2, 'Доставлено с опозданием', sum(delivered_late_cnt)
    FROM mart_delivery_ontime WHERE 1=1 [[AND {{department}}]] [[AND {{report_date}}]]
  UNION ALL
  SELECT 3, 'Отказ', sum(declined_cnt)
    FROM mart_delivery_ontime WHERE 1=1 [[AND {{department}}]] [[AND {{report_date}}]]
  UNION ALL
  SELECT 4, 'В пути', sum(in_transit_cnt)
    FROM mart_delivery_ontime WHERE 1=1 [[AND {{department}}]] [[AND {{report_date}}]]
) x ORDER BY ord;
